# raku-Kris immutable package delta.
# Packages already owned by the pinned Fedora bootc base are intentionally not
# repeated here; the Containerfile protects that base from DNF transactions.
# Keep this list at the feature/root level: Fedora owns the dependency closure.

# Application model
flatpak

# Minimal Plasma shell/session + KWin
plasma-workspace
plasma-desktop
plasma-systemsettings
kwin
kscreen
kscreenlocker

# Plasma Login Manager runtime
plasma-login-manager
kcm-plasmalogin
kde-settings-plasmalogin
kde-settings-plasma
plasma-keyboard
xcb-util-cursor

# Core desktop integration
plasma-nm
plasma-pa
plasma-integration
kde-cli-tools
polkit-kde
powerdevil
kglobalacceld
ksystemstats
xdg-desktop-portal
xdg-desktop-portal-kde
xdg-user-dirs
udisks2

# Network firewall. Fedora 44's plasma-firewall package currently pulls both
# backend plugins; raku-Kris deliberately enables firewalld only. firewalld
# uses the kernel nftables stack and is the supported runtime backend.
plasma-firewall
firewalld

# Mini-PC network + Bluetooth: MediaTek MT7921-class adapter.
# NetworkManager-wifi does not hard-require a Wi-Fi auth backend, so keep
# wpa_supplicant explicit. bluedevil owns the KDE/BlueZ Bluetooth closure.
NetworkManager-wifi
wpa_supplicant
bluedevil
mt7xxx-firmware

# AMD APU / Vega-class graphics: only the hardware-specific firmware plus
# userspace acceleration. Fedora 44's mesa-dri-drivers also provides the
# mesa-va-drivers capability and the radeonsi VA-API backend.
amd-gpu-firmware
amd-ucode-firmware
mesa-dri-drivers
mesa-vulkan-drivers

# Audio/session plumbing. No JACK/pro-audio or Phonon root in M0.
pipewire
pipewire-alsa
pipewire-pulseaudio
wireplumber

# Printing. Keep Fedora's complete CUPS 2.x filter closure because it is used
# for both classic and driverless raster/PCLm conversion. Scanning is provided
# separately by the chosen Epson application, not by a generic immutable SANE
# stack.
plasma-print-manager
cups-filters

# Local desktop/system utilities.
iproute
tar
zram-generator-defaults

# Visual baseline: icons/cursor only. Keep fonts to hard dependencies for M0.
breeze-icon-theme
kf6-breeze-icons
breeze-cursor-theme

# Locales: English + Italian only; never glibc-all-langpacks.
glibc-langpack-en
glibc-langpack-it
langpacks-core-en
langpacks-core-it
