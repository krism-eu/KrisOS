# Network firewall. Fedora 44's plasma-firewall package currently pulls both
# backend plugins; raku-Kris deliberately enables firewalld only. firewalld
# uses the kernel nftables stack and is the supported runtime backend.
plasma-firewall
firewalld

# Mini-PC network + Bluetooth: MediaTek MT7921-class adapter.
# NetworkManager-wifi does not hard-require a Wi-Fi auth backend, so keep
# wpa_supplicant explicit. bluedevil owns the KDE/BlueZ Bluetooth closure.
NetworkManager-wifi
wpa_supplicant
bluedevil
mt7xxx-firmware

# AMD APU / Vega-class graphics: only the hardware-specific firmware plus
# userspace acceleration. Fedora 44's mesa-dri-drivers also provides the
# mesa-va-drivers capability and the radeonsi VA-API backend.
amd-gpu-firmware
amd-ucode-firmware
mesa-dri-drivers
mesa-vulkan-drivers

# Audio/session plumbing. No JACK/pro-audio or Phonon root in M0.
pipewire
pipewire-alsa
pipewire-pulseaudio
wireplumber

# Printing. Keep Fedora's complete CUPS 2.x filter closure because it is used
# for both classic and driverless raster/PCLm conversion. Scanning is provided
# separately by the chosen Epson application, not by a generic immutable SANE
# stack.
plasma-print-manager
cups-filters

# Local desktop/system utilities. NTFS3 is provided by the Fedora kernel;
# ntfs-3g is the userspace fallback and ntfsprogs supplies NTFS maintenance
# tools for the real shared/triboot NTFS partition.
# os-prober is already owned by the pinned Fedora base (1.81-11.fc44 in
# the verified build). Do not repeat it here: the base-collision check would
# reject it. Containerfile asserts its installed package and executable.
iproute
tar
bash-completion
ntfs-3g
ntfsprogs
zram-generator-defaults

# Visual baseline: icons/cursor only. Keep fonts to hard dependencies for M0.
breeze-icon-theme
kf6-breeze-icons
breeze-cursor-theme

# Locales: English + Italian only; never glibc-all-langpacks.
glibc-langpack-en
glibc-langpack-it
langpacks-core-en
langpacks-core-it
